    Required Firestore Security Rules 
    rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() { return request.auth != null; }
    function isAdmin() {
      return isSignedIn() &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }
    match /users/{uid} {
      allow read: if isSignedIn() && (request.auth.uid == uid || isAdmin());
      allow create, update: if isAdmin();
    }
    match /plumberProfiles/{uid} {
      allow read: if isSignedIn() && (request.auth.uid == uid || isAdmin());
      allow create, update: if isAdmin();
    }
    match /claims/{claimId} {
      allow read: if isSignedIn() && (resource.data.plumberUid == request.auth.uid || isAdmin());
      allow create: if isSignedIn() && request.resource.data.plumberUid == request.auth.uid
        && request.resource.data.status == 'Pending';
      allow update: if isAdmin();
    }
    match /redemptions/{redemptionId} {
      allow read: if isSignedIn() && (resource.data.plumberUid == request.auth.uid || isAdmin());
      allow create: if isSignedIn() && request.resource.data.plumberUid == request.auth.uid
        && request.resource.data.status == 'Pending';
      allow update: if isAdmin();
    }
    match /pointsLedger/{entryId} {
      allow read: if isSignedIn() && (resource.data.plumberUid == request.auth.uid || isAdmin());
      allow create: if isSignedIn() && (request.resource.data.plumberUid == request.auth.uid || isAdmin());
    }
    match /schemes/{schemeId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /idRequests/{reqId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }
    match /auditLogs/{logId} {
      allow read, create: if isAdmin();
    }
  }
}
